1. Tách validation bằng FluentValidation và viết trong thư mục Validators
Ví dụ:
public class CreateUserValidator
    : AbstractValidator<CreateUserRequest>
{
    public CreateUserValidator()
    {
        RuleFor(x => x.UserName)
            .NotEmpty()
            .MaximumLength(50);

        RuleFor(x => x.Password)
            .NotEmpty()
            .MinimumLength(8);

        RuleFor(x => x.Email)
            .EmailAddress();

        RuleFor(x => x.Phone)
            .Matches(@"^[0-9]{10}$");
    }
}

2. BusinessRule dùng để validation nhưng cần truy vấn database
ví dụ: kiểm tra user đã tồn tại
private async Task ValidateBusinessAsync(CreateUserRequest request)
{
    if(await _repository.IsUserNameExistAsync(
            request.UserName,
            request.BranchId))
    {
        throw new BusinessException(...);
    }
}
trong service chỉ cần gọi await ValidateBusinessAsync(request);
